-- CAT HUB - Standalone Sea Events Float Boat
-- For a Roblox experience you own.
-- Light-yellow, square UI. No rounded corners.
-- Float height slider: 1-300 studs. Default: 10 studs.

local Players = game:GetService("Players")
local RunService = game:GetService("RunService")
local UserInputService = game:GetService("UserInputService")

local Player = Players.LocalPlayer
local FloatBoat = false
local FloatHeight = 10
local FloatConnection
local AlignPosition
local Attachment

local function GetBoat()
    local Character = Player.Character
    if not Character then return nil end
    local Humanoid = Character:FindFirstChildOfClass("Humanoid")
    if not Humanoid or not Humanoid.SeatPart then return nil end
    return Humanoid.SeatPart:FindFirstAncestorOfClass("Model")
end

local function GetBoatRoot(Boat)
    if not Boat then return nil end
    if Boat.PrimaryPart and Boat.PrimaryPart:IsA("BasePart") then
        return Boat.PrimaryPart
    end
    return Boat:FindFirstChildWhichIsA("BasePart", true)
end

local function StopFloat()
    if FloatConnection then FloatConnection:Disconnect(); FloatConnection = nil end
    if AlignPosition then AlignPosition:Destroy(); AlignPosition = nil end
    if Attachment then Attachment:Destroy(); Attachment = nil end
end

local function StartFloat()
    StopFloat()
    local Boat = GetBoat()
    local Root = GetBoatRoot(Boat)
    if not Root or Root.Anchored then return false end

    Attachment = Instance.new("Attachment")
    Attachment.Name = "CAT_HUB_FloatAttachment"
    Attachment.Parent = Root

    AlignPosition = Instance.new("AlignPosition")
    AlignPosition.Name = "CAT_HUB_FloatPosition"
    AlignPosition.Mode = Enum.PositionAlignmentMode.OneAttachment
    AlignPosition.Attachment0 = Attachment
    AlignPosition.MaxForce = math.huge
    AlignPosition.MaxVelocity = math.huge
    AlignPosition.Responsiveness = 20
    AlignPosition.RigidityEnabled = false
    AlignPosition.Parent = Root

    local BaseY = Root.Position.Y
    FloatConnection = RunService.Heartbeat:Connect(function()
        if not FloatBoat or not Root.Parent then return end
        AlignPosition.Position = Vector3.new(Root.Position.X, BaseY + FloatHeight, Root.Position.Z)
    end)
    return true
end

local function SetFloatBoat(Value)
    FloatBoat = Value
    if Value then StartFloat() else StopFloat() end
end

-- UI
local Gui = Instance.new("ScreenGui")
Gui.Name = "CAT_HUB_SeaEvents"
Gui.ResetOnSpawn = false
Gui.Parent = Player:WaitForChild("PlayerGui")

local Main = Instance.new("Frame")
Main.Name = "SeaEvents"
Main.Size = UDim2.fromOffset(320, 180)
Main.Position = UDim2.new(0.5, -160, 0.72, 0)
Main.BackgroundColor3 = Color3.fromRGB(255, 242, 170)
Main.BorderSizePixel = 1
Main.Parent = Gui

local Title = Instance.new("TextLabel")
Title.BackgroundTransparency = 1
Title.Position = UDim2.fromOffset(10, 8)
Title.Size = UDim2.new(1, -20, 0, 30)
Title.Font = Enum.Font.GothamBold
Title.Text = "CAT HUB  •  Sea Events"
Title.TextColor3 = Color3.fromRGB(55, 45, 20)
Title.TextSize = 20
Title.Parent = Main

local Toggle = Instance.new("TextButton")
Toggle.Position = UDim2.fromOffset(10, 46)
Toggle.Size = UDim2.new(1, -20, 0, 38)
Toggle.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
Toggle.BorderSizePixel = 1
Toggle.Font = Enum.Font.GothamBold
Toggle.Text = "Float Boat: OFF"
Toggle.TextColor3 = Color3.fromRGB(55, 45, 20)
Toggle.TextSize = 16
Toggle.Parent = Main

local HeightLabel = Instance.new("TextLabel")
HeightLabel.BackgroundTransparency = 1
HeightLabel.Position = UDim2.fromOffset(10, 91)
HeightLabel.Size = UDim2.new(1, -20, 0, 24)
HeightLabel.Font = Enum.Font.Gotham
HeightLabel.Text = "Float Height: 10 studs"
HeightLabel.TextColor3 = Color3.fromRGB(55, 45, 20)
HeightLabel.TextSize = 15
HeightLabel.Parent = Main

local SliderBar = Instance.new("Frame")
SliderBar.Position = UDim2.fromOffset(18, 124)
SliderBar.Size = UDim2.new(1, -36, 0, 8)
SliderBar.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
SliderBar.BorderSizePixel = 1
SliderBar.Parent = Main

local SliderFill = Instance.new("Frame")
SliderFill.Size = UDim2.new((FloatHeight - 1) / 49, 0, 1, 0)
SliderFill.BackgroundColor3 = Color3.fromRGB(245, 190, 45)
SliderFill.BorderSizePixel = 0
SliderFill.Parent = SliderBar

local SliderKnob = Instance.new("TextButton")
SliderKnob.AnchorPoint = Vector2.new(0.5, 0.5)
SliderKnob.Size = UDim2.fromOffset(18, 18)
SliderKnob.Position = UDim2.new((FloatHeight - 1) / 49, 0, 0.5, 0)
SliderKnob.BackgroundColor3 = Color3.fromRGB(245, 190, 45)
SliderKnob.BorderSizePixel = 1
SliderKnob.Text = ""
SliderKnob.AutoButtonColor = false
SliderKnob.Parent = SliderBar

local dragging = false
local function UpdateHeight(x)
    local alpha = math.clamp((x - SliderBar.AbsolutePosition.X) / SliderBar.AbsoluteSize.X, 0, 1)
    FloatHeight = math.floor(1 + alpha * 49 + 0.5)
    local normalized = (FloatHeight - 1) / 49
    SliderFill.Size = UDim2.new(normalized, 0, 1, 0)
    SliderKnob.Position = UDim2.new(normalized, 0, 0.5, 0)
    HeightLabel.Text = "Float Height: " .. FloatHeight .. " studs"
    if FloatBoat then StartFloat() end
end

local function BeginDrag(Input)
    if Input.UserInputType == Enum.UserInputType.MouseButton1 or Input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        UpdateHeight(Input.Position.X)
    end
end

SliderBar.InputBegan:Connect(BeginDrag)
SliderKnob.InputBegan:Connect(BeginDrag)
UserInputService.InputChanged:Connect(function(Input)
    if dragging and (Input.UserInputType == Enum.UserInputType.MouseMovement or Input.UserInputType == Enum.UserInputType.Touch) then
        UpdateHeight(Input.Position.X)
    end
end)
UserInputService.InputEnded:Connect(function(Input)
    if Input.UserInputType == Enum.UserInputType.MouseButton1 or Input.UserInputType == Enum.UserInputType.Touch then
        dragging = false
    end
end)

Toggle.MouseButton1Click:Connect(function()
    SetFloatBoat(not FloatBoat)
    Toggle.Text = "Float Boat: " .. (FloatBoat and "ON" or "OFF")
end)

Player.CharacterRemoving:Connect(StopFloat)
