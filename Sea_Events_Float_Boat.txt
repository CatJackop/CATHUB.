-- CAT HUB - Standalone Sea Events Float Boat
-- For a Roblox experience you own.
-- UI: Sea Events -> Float Boat ON/OFF + Height slider
--
-- Setup:
-- 1. Put this LocalScript in StarterPlayerScripts.
-- 2. Your boat should be a Model containing a BasePart or VehicleSeat.
-- 3. Set the boat's PrimaryPart if possible.
--
-- The script uses AlignPosition to hold the boat at the selected
-- height above the current water level. It does not use game-specific
-- remotes or bypass server mechanics.

local Players = game:GetService("Players")
local RunService = game:GetService("RunService")

local player = Players.LocalPlayer

local floatEnabled = false
local floatHeight = 8
local floatAttachment
local alignPosition
local heartbeatConnection

local function getBoat()
    local character = player.Character
    if not character then
        return nil
    end

    local seat = character:FindFirstChildOfClass("Humanoid")
        and character:FindFirstChildOfClass("Humanoid").SeatPart

    if not seat then
        return nil
    end

    local boat = seat:FindFirstAncestorOfClass("Model")
    if not boat then
        return nil
    end

    return boat
end

local function getRootPart(boat)
    if boat.PrimaryPart then
        return boat.PrimaryPart
    end

    return boat:FindFirstChildWhichIsA("BasePart", true)
end

local function stopFloating()
    if heartbeatConnection then
        heartbeatConnection:Disconnect()
        heartbeatConnection = nil
    end

    if alignPosition then
        alignPosition:Destroy()
        alignPosition = nil
    end

    if floatAttachment then
        floatAttachment:Destroy()
        floatAttachment = nil
    end
end

local function startFloating()
    stopFloating()

    local boat = getBoat()
    if not boat then
        warn("Float Boat: sit in your boat first.")
        return
    end

    local root = getRootPart(boat)
    if not root or root.Anchored then
        warn("Float Boat: no usable unanchored boat part found.")
        return
    end

    floatAttachment = Instance.new("Attachment")
    floatAttachment.Name = "CAT_HUB_FloatAttachment"
    floatAttachment.Parent = root

    alignPosition = Instance.new("AlignPosition")
    alignPosition.Name = "CAT_HUB_FloatPosition"
    alignPosition.Mode = Enum.PositionAlignmentMode.OneAttachment
    alignPosition.Attachment0 = floatAttachment
    alignPosition.MaxForce = math.huge
    alignPosition.MaxVelocity = math.huge
    alignPosition.Responsiveness = 20
    alignPosition.RigidityEnabled = false
    alignPosition.Parent = root

    heartbeatConnection = RunService.Heartbeat:Connect(function()
        if not floatEnabled then
            return
        end

        if not root.Parent or not root:IsDescendantOf(workspace) then
            stopFloating()
            return
        end

        -- Keep the boat at its current X/Z while maintaining its
        -- current Y position plus the requested float height.
        local targetY = root.Position.Y + floatHeight
        alignPosition.Position = Vector3.new(
            root.Position.X,
            targetY,
            root.Position.Z
        )
    end)
end

local function setFloatEnabled(value)
    floatEnabled = value

    if floatEnabled then
        startFloating()
    else
        stopFloating()
    end
end

-- Minimal standalone UI.
local gui = Instance.new("ScreenGui")
gui.Name = "CAT_HUB_SeaEvents"
gui.ResetOnSpawn = false
gui.Parent = player:WaitForChild("PlayerGui")

local panel = Instance.new("Frame")
panel.Name = "SeaEvents"
panel.Size = UDim2.fromOffset(300, 150)
panel.Position = UDim2.new(0.5, -150, 0.75, 0)
panel.BackgroundColor3 = Color3.fromRGB(255, 242, 170)
panel.BorderSizePixel = 0
panel.Parent = gui

local corner = Instance.new("UICorner")
corner.CornerRadius = UDim.new(0, 12)
corner.Parent = panel

local title = Instance.new("TextLabel")
title.Size = UDim2.new(1, -20, 0, 32)
title.Position = UDim2.fromOffset(10, 8)
title.BackgroundTransparency = 1
title.Text = "Sea Events"
title.TextSize = 20
title.Font = Enum.Font.GothamBold
title.TextColor3 = Color3.fromRGB(50, 45, 20)
title.Parent = panel

local toggle = Instance.new("TextButton")
toggle.Size = UDim2.new(1, -20, 0, 36)
toggle.Position = UDim2.fromOffset(10, 48)
toggle.Text = "Float Boat: OFF"
toggle.TextSize = 16
toggle.Font = Enum.Font.GothamBold
toggle.TextColor3 = Color3.fromRGB(50, 45, 20)
toggle.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
toggle.Parent = panel

local toggleCorner = Instance.new("UICorner")
toggleCorner.CornerRadius = UDim.new(0, 8)
toggleCorner.Parent = toggle

local slider = Instance.new("TextBox")
slider.Size = UDim2.new(1, -20, 0, 36)
slider.Position = UDim2.fromOffset(10, 92)
slider.Text = "Float Height: 8"
slider.PlaceholderText = "Height (1-50)"
slider.TextSize = 15
slider.Font = Enum.Font.Gotham
slider.TextColor3 = Color3.fromRGB(50, 45, 20)
slider.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
slider.ClearTextOnFocus = false
slider.Parent = panel

local sliderCorner = Instance.new("UICorner")
sliderCorner.CornerRadius = UDim.new(0, 8)
sliderCorner.Parent = slider

toggle.MouseButton1Click:Connect(function()
    setFloatEnabled(not floatEnabled)
    toggle.Text = "Float Boat: " .. (floatEnabled and "ON" or "OFF")

    if floatEnabled then
        -- Reacquire the boat after the player sits in it.
        task.delay(0.2, function()
            if floatEnabled then
                startFloating()
            end
        end)
    end
end)

slider.FocusLost:Connect(function()
    local value = tonumber(slider.Text:match("%-?%d+%.?%d*"))

    if value then
        floatHeight = math.clamp(value, 1, 50)
    end

    slider.Text = "Float Height: " .. tostring(floatHeight)

    if floatEnabled then
        startFloating()
    end
end)

player.CharacterRemoving:Connect(function()
    stopFloating()
end)
